@Service
public class DebtSettlementService {

    public List<SettlementTransaction> simplifyDebts(Long groupId) {
        Map<User, BigDecimal> balances = calculateNetBalances(groupId);
        
        PriorityQueue<BalanceNode> debtors = new PriorityQueue<>(Comparator.comparing(b -> b.amount));
        PriorityQueue<BalanceNode> creditors = new PriorityQueue<>((a, b) -> b.amount.compareTo(a.amount));
        
        // Populate max-heap & min-heap for greedy debt settlement
        balances.forEach((user, balance) -> {
            if (balance.compareTo(BigDecimal.ZERO) < 0) 
                debtors.add(new BalanceNode(user, balance.abs()));
            else if (balance.compareTo(BigDecimal.ZERO) > 0) 
                creditors.add(new BalanceNode(user, balance));
        });

        List<SettlementTransaction> transactions = new ArrayList<>();
        while (!debtors.isEmpty() && !creditors.isEmpty()) {
            BalanceNode debtor = debtors.poll();
            BalanceNode creditor = creditors.poll();
            BigDecimal min = debtor.amount.min(creditor.amount);
            
            transactions.add(new SettlementTransaction(debtor.user, creditor.user, min));
            
            if (debtor.amount.compareTo(min) > 0) 
                debtors.add(new BalanceNode(debtor.user, debtor.amount.subtract(min)));
            if (creditor.amount.compareTo(min) > 0) 
                creditors.add(new BalanceNode(creditor.user, creditor.amount.subtract(min)));
        }
        return transactions; // Max n-1 transactions
    }
}
